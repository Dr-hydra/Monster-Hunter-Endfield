cmake_minimum_required(VERSION 3.25)
project(MonsterHunterEndfield VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "First release supports Windows x64")
endif()
set(SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/include" CACHE PATH "E Mod Loader SDK and JSON headers")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/module.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/module.json" MHE_MANIFEST_JSON)
string(JSON MHE_PACKAGE_VERSION GET "${MHE_MANIFEST_JSON}" version)
if(NOT MHE_PACKAGE_VERSION MATCHES "^[0-9A-Za-z.+-]+$")
 message(FATAL_ERROR "Invalid module package version")
endif()
add_library(longsword SHARED native/module.cpp native/build_identity.cpp)
target_compile_definitions(longsword PRIVATE "MHE_PACKAGE_VERSION=\"${MHE_PACKAGE_VERSION}\"")
target_include_directories(longsword PRIVATE "${SDK_INCLUDE}")
target_compile_definitions(longsword PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(longsword PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(longsword PRIVATE user32 winmm gdi32)
set(MHE_MODULE_EXTENSION "" CACHE FILEPATH "Optional local module extension CMake file")
if(MHE_MODULE_EXTENSION)
 get_filename_component(MHE_MODULE_EXTENSION "${MHE_MODULE_EXTENSION}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
 if(NOT EXISTS "${MHE_MODULE_EXTENSION}")
  message(FATAL_ERROR "The requested local module extension is unavailable")
 endif()
 include("${MHE_MODULE_EXTENSION}")
 mhe_attach_module_extension(longsword)
endif()
set(PACKAGE_DIR "${CMAKE_BINARY_DIR}/package")
set_target_properties(longsword PROPERTIES OUTPUT_NAME "endfield.longsword.showcase")
foreach(CONFIG RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
 set_target_properties(longsword PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${PACKAGE_DIR}/native/windows-x64")
endforeach()
