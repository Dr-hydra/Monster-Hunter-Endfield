cmake_minimum_required(VERSION 3.25)
project(EndfieldLongswordShowcase VERSION 0.5.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "First release supports Windows x64")
endif()
set(SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/include" CACHE PATH "E Mod Loader SDK and JSON headers")
add_library(longsword SHARED native/module.cpp)
target_include_directories(longsword PRIVATE "${SDK_INCLUDE}")
target_compile_definitions(longsword PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(longsword PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(longsword PRIVATE user32 winmm gdi32)
set(PACKAGE_DIR "${CMAKE_BINARY_DIR}/package")
set_target_properties(longsword PROPERTIES OUTPUT_NAME "endfield.longsword.showcase")
foreach(CONFIG RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
 set_target_properties(longsword PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${PACKAGE_DIR}/native/windows-x64")
endforeach()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_compile_options(core_tests PRIVATE /W4 /utf-8 /permissive-)
add_test(NAME longsword_core COMMAND core_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets")
add_executable(runtime_tests tests/runtime_tests.cpp)
target_include_directories(runtime_tests PRIVATE "${SDK_INCLUDE}")
target_compile_definitions(runtime_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(runtime_tests PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(runtime_tests PRIVATE user32 winmm gdi32)
add_test(NAME longsword_runtime COMMAND runtime_tests "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(audio_tests tests/audio_tests.cpp)
target_compile_definitions(audio_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(audio_tests PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(audio_tests PRIVATE winmm)
add_test(NAME longsword_audio COMMAND audio_tests "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(retarget_tests tests/retarget_tests.cpp)
target_include_directories(retarget_tests PRIVATE "${SDK_INCLUDE}")
target_compile_options(retarget_tests PRIVATE /W4 /utf-8 /permissive-)
add_test(NAME longsword_retarget COMMAND retarget_tests "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(practice_tests tests/practice_tests.cpp)
target_compile_options(practice_tests PRIVATE /W4 /utf-8 /permissive-)
add_test(NAME longsword_practice COMMAND practice_tests)
add_executable(weapon_tests tests/weapon_tests.cpp)
target_compile_options(weapon_tests PRIVATE /W4 /utf-8 /permissive-)
add_test(NAME longsword_weapon COMMAND weapon_tests)
